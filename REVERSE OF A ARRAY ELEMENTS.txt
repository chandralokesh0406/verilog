REVERSE OF A ARRAY ELEMENTS:

module reverse_a_array_elemets(clk,reset,arr,arr_1);
  input clk,reset;
  input [7:0]arr[0:4];
  output  reg [7:0]arr_1[0:4];
  
  parameter size = 5;
  int i;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      arr_1 <= '{0,0,0,0,0};
    else
      begin
        for( i=0; i<size; i++) begin
        arr_1[size-1-i] <= arr[i];
        end
      end
  end
endmodule
