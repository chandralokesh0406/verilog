Count numbers divisible by X between two ranges:


module counting_numbers_divible_in_ranges(clk,reset,x,num1,num2,out);
  input clk,reset;
  input [15:0]x;
  input [31:0]num1,num2;
  output reg [31:0]out;
  
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  //parameter s2 = 2'b10;
  
  reg [31:0]num,digit;
  
  reg [1:0]current_state,next_state;
  
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          if( num < digit)
            next_state = s1;
          else
            next_state = s0;
        end
    endcase
  end
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      current_state <= s0;
      num <= num1;
      digit <= num2;
      out <= 0;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
          s0:
            begin
              num <= num1;
              digit <= num2;
              out <= 0;
            end
          s1:
            begin
              if( num % x == 0) begin
                out <= out + 1;
                num <= num + 1;
              end
              else
                num <= num+1;
            end
        endcase
      end
  end
endmodule
