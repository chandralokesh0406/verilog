counting_numbers_any_two_tables:
module counting_numbers_any_two_tables(clk,reset,num1,num2,count);
  input clk,reset;
  input [7:0]num1,num2;
  output reg [15:0]count;
  
  reg [7:0] digit1,digit2;
  
  reg [15:0]i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      i <= 1;
      digit1 <= num1;
      digit2 <= num2;
      count <= 0;
    end
    else if( i % digit1 == 0 && i % digit2 == 0) begin
      count <= count+1;
      i <= i+1;
    end
    else
      i <= i+1;
  end
endmodule
