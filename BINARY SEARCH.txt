module linear_search(clk,reset,arr,num,found);
  input clk,reset;
  input [7:0] arr [0:4];
  input [7:0] num;
  output reg found;
  parameter size = 5;
  
  reg [2:0]i;
  always @(posedge clk or negedge reset) begin
    
    if(!reset)
      found <= 0;
    else begin
      found <= 0;
      for(i=0; i<size; i=i+1) begin
        if(num == arr[i])
          found <= 1;
      end
    end
  end
endmodule
