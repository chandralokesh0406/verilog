module checking_the_number_is_in_array_or_not(clk,reset,arr,num,out);
  input clk,reset;
  input [7:0]arr[0:4];
  input [7:0]num;
  output reg out;
  
  
  parameter size = 5;
  int i;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
    end
    else
      begin
        out <= 0;
        for(i=0; i<size; i++) begin
          if(arr[i] == num)
            out <= 1;
        end
      end
  end
endmodule
