PROBLEM STATMENT:

🧾 20. Find Missing Digit in Sequence
🔹 Objective

Find missing number in sequence

🔍 Example

1,2,3,5

🎯 Output

👉 4


design:

module finding_missing_digit_in_a_sequence(clk,reset,num,out);
  input clk,reset;
  input [3:0]num[0:3];
  output reg [3:0]out;
  
  int i;
  reg [3:0]count;
  
  always @(posedge clk or negedge reset)
    begin
      if(!reset) begin
        out <= 0;
        i  <= 0;
        count <= 1;
      end
      else
        begin
          for( i=0; i<$size(num); i++) begin
            if(count != num[i])
              out = count;
            else if(out != count)
            count = count + 1;
          end
        end
    end
endmodule
