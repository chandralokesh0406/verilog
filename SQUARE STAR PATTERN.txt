SQUARE WAVE PATTERN

module square_star_pattern (
    input clk,
    input reset
);

    reg [1:0] row;
    reg [1:0] col;

    parameter SIZE = 4;

    always @(posedge clk or negedge reset) begin

        if(!reset) begin
            row <= 0;
            col <= 0;
        end

        else begin

            if(row < SIZE) begin

                if(col < SIZE-1) begin
                  $write(" * ");
                    col <= col + 1;
                end

                else begin
                  $write(" * ");
                    $display("");
                    col <= 0;
                    row <= row + 1;
                end

            end

        end

    end

endmodule
