REVERSE A STRING:

module reverse_a_string(clk,reset,str,rev_str);
  input clk,reset;
  input string str;
  output string rev_str;
  
  parameter size = 10;
  
  
  int i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      rev_str <= {size{"0"}};
    else
      for(i=0; i<size; i++) begin
        rev_str[size-1-i] <= str[i];
      end
  end
endmodule
