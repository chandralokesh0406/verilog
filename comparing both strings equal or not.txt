module comparing_the_both_strings_same_or_not(clk,reset,sr_1,sr_2,out);
  input clk,reset;
  input string sr_1;
  input string sr_2;
  output reg out;
  
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 0;
    else
      begin
        if( sr_1 == sr_2 )
          out <= 1;
        else
          out <= 0;
      end
  end
endmodule
