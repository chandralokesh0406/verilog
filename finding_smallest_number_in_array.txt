module finding_smallest_number_in_an_array(clk,reset,arr,out);
  input clk,reset;
  input [7:0]arr[4:0];
  output reg [7:0]out;
  
  
  reg [7:0]min;
  
  parameter size = 5;
  
  int i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 0;
    else
      begin
        min = arr[0];
        for(i = 1; i<size; i++) begin
          if( arr[i] < min)
            min = arr[i];
        end
        out <= min;
      end
  end
endmodule
