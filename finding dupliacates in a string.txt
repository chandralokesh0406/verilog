module finding_duplicates_in_a_string(clk,reset,sr_1,duplication);
  input clk,reset;
  input string sr_1;
  output  string duplication;
  
  bit found;
  

  
  string temp_duplication;
  int i;
  int k;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      duplication <= "";
      i <= 0;
      found <= 0;
    end
    else begin
      if(i < sr_1.len()) begin
        for(int j = sr_1.len()-1; j > i; j--) begin
          if( sr_1[j] == sr_1[i]) begin
            found = 0;
            for(k=0; k<duplication.len(); k++) begin
              if(duplication[k] == sr_1[j])
                found = 1;
            end
            if(!found)
            duplication <= {duplication,sr_1[j]};
          end
        end
        i<= i+1;
      end
    end
  end
endmodule
