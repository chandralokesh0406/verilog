🧾 Problem Summary: ASCII Difference of String
🔹 Objective

Given a string input, compute:

The difference between ASCII values of adjacent characters

Then sum all those differences

Store the result in out


design code:
module problem_statement(clk,reset,string_,out);
  input clk,reset;
  input string string_;
  output reg[7:0]out;

  
  int i;
  reg [7:0]diff;

  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      diff <= 0;
    end
    else
      begin
        if(i < string_.len()) begin
          diff <= $signed(string_[i]) - $signed(string_[i+1]);
          out <= (out + diff);
        end
        i<= i+1;
      end
  end
endmodule
