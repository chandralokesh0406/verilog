module problem_statement(clk,reset,num,out);
  input clk,reset;
  input [15:0]num;
  output reg [7:0]out;
  
  
  reg [7:0]small_,smallest_number,greatest_number,sum,sum_;
  reg [15:0]digit,digit_1;
  
  
  always @(*) begin
    digit = num;
    digit_1 = num;
    smallest_number = digit % 10;
    small_ = 0;
    while (digit_1 > 0) begin
      digit = digit_1 % 10;
      digit_1 = digit_1/10;
      small_ = digit;
      if(small_ > smallest_number)
        greatest_number = small_;
      else
        smallest_number = small_;
    end
  end
  
  always @(*) begin
    sum = greatest_number + smallest_number;
  end
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      sum_ <= 0;
    end
    else
      begin
        for(int i= 0; i<sum+1; i++) begin
          if(sum % i == 0)
            sum_ = sum_ + i;
        end
        out <= sum_;
        sum_<= 0;
      end
  end
endmodule
