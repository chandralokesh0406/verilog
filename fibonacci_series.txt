FIBONACCI SERIES:


module fibonacci_series(clk,reset,out);
  input wire clk,reset;
  output reg [31:0]out;
  
  
  reg [31:0]a,b;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      a <= 0;
      b <= 1;
      out <= 0;
    end
    else
      begin
        out <= a+b;
        a <= b;
        b <=a+b;  here when we decalare out in place of a+b it takes previous olda values
      end
  end
endmodule
