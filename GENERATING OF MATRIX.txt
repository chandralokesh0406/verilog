module matrix(
    input clk,
    input reset,
    output reg [7:0] mat [0:1][0:1]
);

always @(posedge clk or negedge reset) begin
    if(!reset) begin
        mat[0][0] <= 0;
        mat[0][1] <= 0;
        mat[1][0] <= 0;
        mat[1][1] <= 0;
    end
    else begin
        mat[0][0] <= 8'd1;
        mat[0][1] <= 8'd2;
        mat[1][0] <= 8'd3;
        mat[1][1] <= 8'd4;
    end
end

endmodule
