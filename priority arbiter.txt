module priority_arbiter(clk,reset,i,q);

  input clk,reset;
  input [3:0]i;

  output reg [3:0]q;

  always @(posedge clk or negedge reset) begin

    if(!reset)
      q = 4'b0000;

    else begin

      if(i[3] == 1)
        q[3] <= 1;

      else if(i[2] == 1)
        q[2] <= 1;

      else if(i[1] == 1)
        q[1] <= 1;

      else
        q[0] <= 1;

    end

  end

endmodule
