ROUND ROBIN ARIBITER:

module round_robin_aribiter(clk,reset,m,grant);
  input clk,reset;
  input [3:0]m;
  output reg [3:0]grant;
  
  reg [1:0]ptr;
  
  
  always @(*) begin
    grant = 4'b0000;
    case(ptr)
      2'b00:
        begin
          if(m[0])
            grant = 4'b0001;
          else if(m[1])
            grant = 4'b0010;
          else if(m[2])
            grant = 4'b0100;
          else if(m[3])
            grant = 4'b1000;
        end
      2'b01:
        begin
          if(m[1])
            grant = 4'b0010;
          else if(m[2])
            grant = 4'b0100;
          else if(m[3])
            grant = 4'b1000;
          else if(m[0])
            grant = 4'b0001;
        end
      2'b10:
        begin
          if(m[2])
            grant = 4'b0100;
          else if(m[3])
            grant = 4'b1000;
          else if(m[0])
            grant = 4'b0001;
          else if(m[1])
            grant = 4'b0010;
        end
      2'b11:
        begin
          if(m[3])
            grant = 4'b1000;
          else if(m[0])
            grant = 4'b0001;
          else if(m[1])
            grant = 4'b0010;
          else if(m[2])
            grant = 4'b0100;
        end
      default: grant = 4'b0000;
    endcase
  end
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      ptr <= 0;
    else
      case(grant)
        4'b0001: ptr <= 2'b01;
        4'b0010: ptr <= 2'b10;
        4'b0100: ptr <= 2'b11;
        4'b1000: ptr <= 2'b00;
        default: ptr <= ptr;
      endcase
  end
endmodule

    end

    end
endmodule
