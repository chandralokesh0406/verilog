module stack(clk,reset,wr_en,rd_en,data_in,data_out);
  input clk,reset;
  input wr_en;
  input rd_en;
  input [7:0]data_in;
  output reg [7:0]data_out;
  
  reg full,empty;
  reg [7:0]stack[0:15];
  
  reg [4:0]sp;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      data_out <= 0;
      full <= 0;
      empty <= 1;
      sp <= 0;
    end
    else
      begin
        if(wr_en == 1 && full !=1) begin
          stack[sp] <= data_in;
          sp <= sp + 1;
          empty <= 0;
          if(sp+1 == 16)
            full <= 1;
        end
        else if(rd_en == 1 && empty == 0)
          begin
            data_out <= stack[sp-1];
            sp <= sp - 1;
            full <= 0;
            if(sp == 1)
              empty <= 1;
          end
      end
  end
endmodule
