module sum_of_digits_prime_or_not(clk,reset,num,prime);
  input clk,reset;
  input [7:0]num;
  output reg prime;
  
  reg [1:0]next_state,current_state;
  reg [7:0]digit,digit_1,out;
  reg [3:0]i;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  parameter s3 = 2'b11;
  
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          if(digit_1 > 0)
            next_state = s1;
          else
            next_state = s2;
        end
      s2:
        begin
          next_state = s0;
        end
    endcase
  end
  
  
  
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      current_state <= s0;
      prime <= 1;
      digit <= 0;
      digit_1 <= num;
      out <= 0;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
          s0:
            begin
              prime <= 1;
              digit <= 0;
              digit_1 <= num;
              out <= 0;
            end
          s1:
            begin
              if(digit_1 > 0) begin
                digit_1 <= digit_1 / 10;
                out <= out + (digit_1 % 10);
              end
            end
          s2:
            begin
              for( i=2; i<out; i++) begin
                if(out % i == 0)
                  prime <= 0;
              end
            end
        endcase
      end
  end
endmodule
