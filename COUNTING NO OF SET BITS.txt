module counting_set_bits(clk,reset,in,out);
  input clk,reset;
  input [3:0]in;
  output reg [2:0]out;
  
  reg [2:0]dec,width,temp;
  reg [3:0]shift;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  
  
  reg[1:0]current_state,next_state;
  
  always @(*) begin
    width = 0;
    temp = in;
    while (temp > 0) begin
      temp = temp >> 1;
      width = width+1;
    end
  end
  
    
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          if(dec > 0)
            next_state = s1;
          else
            next_state = s0;
        end
    endcase
  end
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      shift <= 0;
      current_state <= s0;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
          s0:
            begin
              out <= 0;
              shift <= in;
              dec <= width;
            end
          s1:
            begin
              shift <= {shift >> 1};
              if(shift[0] == 1) begin
                out <= out + 1;
                dec <= dec - 1;
              end
              else
                dec <= dec - 1;
            end
        endcase
      end
  end
endmodule
