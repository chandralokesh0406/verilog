module parallelogram_star_printing(clk,reset);
  input clk,reset;
  
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  parameter s3 = 2'b11;
  
  parameter size = 5;
  
  reg [1:0]current_state,next_state;
  
  reg [2:0]row,col,space;
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          if(space < row)
            next_state = s1;
          else
            next_state = s2;
        end
      s2:
        begin
          if(col < size)
            next_state = s2;
          else
            next_state = s3;
        end
      s3:
        begin
          next_state = s1;
        end
    endcase
  end
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      row <= 0;
      col <= 0;
      space <= 0;
      current_state <= s0;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
          s0:
            begin
              col <= 0;
              space <= 0;
            end
          s1:
            begin
              if(space < row)
                begin
                  $write(" ");
                  space <= space + 1;
                end
            end
          s2:
            begin
              if(col < size)begin
                $write("*");
                col <= col+1;
              end
            end
          s3:
            begin
              $display("");
              row <= row + 1;
              col <= 0;
              space <= 0;
              
              if(row == size - 1)
                $finish;
            end
        endcase
      end
  end
endmodule
