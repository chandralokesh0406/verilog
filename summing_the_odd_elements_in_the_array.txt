module summing_the_odd_elements_in_the_array(clk,reset,arr,out);
  input clk,reset;
  input [7:0]arr[0:4];
  output reg [7:0]out;
  
  parameter size = 5;
  int i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 0;
    else
      begin
        if( out == 0)
          for(i=0; i<size; i++) begin
            if(i%2 != 0)
              out = out + arr[i];
          end
        else
          out <= 0;
      end
  end
endmodule
