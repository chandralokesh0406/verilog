FINDING LARGEST ELEMENT IN AN ARRAY:

module finding_largest_number_in_an_array(clk,reset,arr,out);
  input clk,reset;
  input  [7:0]arr[4:0];
  output reg [7:0]out;
  parameter size = 5;
  

  reg [7:0]max;
  
  int i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 0;
    else
      begin
        max = arr[0];
        for ( i = 1; i < size; i++) begin
          if(arr[i] > max)
            max = arr[i];
        end
        out <= max;
      end
  end
endmodule
