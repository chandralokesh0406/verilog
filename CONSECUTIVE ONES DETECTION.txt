CONSECUTIVE ONES DETECTION:

module consecutive_ones_detection(clk,reset,in,out);
  input clk,reset;
  input [7:0]in;
  output reg out;
  
  reg [2:0]i;
  
  parameter size = 8;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      begin
        out <= 0;
        i <= 0;
      end
    else
      begin
        if(in[i] && in[i+1]) begin
          out <= 1;
          i <= i+1;
        end
        else
          begin
            out <= 0;
            i <= i+1;
            if(i== size - 2)
              i <= 0;
          end
      end
  end
endmodule
