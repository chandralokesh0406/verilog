SUM OF DIGITS IN A STRING:


module sum_of_the_string(clk,reset,chandu,out);
  input clk,reset;
  input string chandu;
  output reg [7:0]out;
  
  reg [7:0]result;
  
  parameter size = 3;
  
  int i;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      result <= 0;
    end
    else
      begin
        for(i=0; i<size; i++) begin
          result = result+( chandu[i] - "0");
        end
        out <= result;
      end
  end
endmodule
