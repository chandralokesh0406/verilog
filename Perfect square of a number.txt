Perfect square of a number:

module finding_perfect_square(clk,reset,num,out);
  input clk,reset;
  input [127:0]num;
  output reg out;
  
  reg [127:0]digit;
  
  reg [127:0]i;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  
  reg [1:0]current_state , next_state;
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      
      s1:
        begin
          if(i*i >= digit)
            next_state = s2;
          else
            next_state = s1;
        end
      s2:
        begin
          next_state = s0;
        end
    endcase
  end
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      current_state <= s0;
      digit <= num;
      out <= 0;
      i <= 0;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
          s0:
            begin
              digit <= num;
              out <= 0;
              i <= 0;
            end
          s1:
            begin
              if( i*i == digit)
                out <= 1;
              else
                i <= i+1;
            end
          s2:
            begin
            end
        endcase
      end
  end
endmodule
