
CHECKING THE DIGITS OR CHARECTERS IN A STRING:

module checking_the_digits_or_charecters_in_a_string_or_not(clk,reset,in,charecter,out);
  input clk,reset;
  input string in;
  input string charecter;
  output reg [7:0]out;
  
  int i;
  string  char;
  
  parameter size = 15;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      char <= charecter;
    end
    else
      begin
        for(i=0; i<size; i++) begin
          if(char == in[i]) begin
            out <= 1;
          end
        end
      end
  end
endmodule
