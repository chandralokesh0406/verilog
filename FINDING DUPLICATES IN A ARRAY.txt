FINDING DUPLICATES IN A ARRAY:

module finding_duplicates_in_a_array(clk,reset,arr,duplication);
  input clk,reset;
  input [7:0]arr[0:4];
  output reg [7:0]duplication[0:4];
  
  bit found;
  
  int i;
  int j;
  int k;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      duplication <= '{0,0,0,0,0};
      i <= 0;
      found = 0;
    end
    else
      begin
        if(i<$size(arr))
          for(j=$size(arr)-1; j>i; j--) begin
          if(arr[j] == arr[i]) begin
            found = 0;
            for(k=0; k<$size(duplication); k++) begin
              if(duplication[k] == arr[j])
                found = 1;
            end
            if(!found)
              duplication[i] <= arr[j];
          end
        end
        i<= i+1;
      end
  end
endmodule
