module summing_the_odd_elements_in_the_array(clk,reset,arr,out);
  input clk,reset;
  input [7:0]arr[0:4];
  output reg [7:0]out;
  
  parameter size = 5;
  int i;
  reg [7:0]temp_sum;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      temp_sum <= 0;
    end
    else
      begin
        for(i=0; i<size; i++) begin
          if(i%2!=0)
            temp_sum = temp_sum+arr[i];
        end
        out = temp_sum;
        temp_sum <= 0;
      end
  end
endmodule
