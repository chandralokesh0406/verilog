CHECKING ODD PARITY:

even no of ones p = 1;
odd no of ones p = 0;

module checking_odd_parity(clk,reset,bits,p);
  input clk,reset;
  input [3:0]bits;
  output reg p;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  
  reg [3:0]count;
  reg [2:0]i;
  parameter size = 5;
  
  reg [1:0]current_state,next_state;
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          next_state = s2;
        end
      s2:
        begin
          next_state = s0;
        end
    endcase
  end
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      p <= 0;
      count <= 0;
      current_state <= s0;
    end
    else begin
      current_state <= next_state;
      case(current_state)
        s0:
          begin
            p <= 0;
            count <= 0;
          end
        s1:
          begin
            for(i=0; i<size; i++) begin
              if(bits[i] == 1)
                count = count+1;
            end
          end
        s2:
          begin
            if(count % 2 == 0)
              p <= 1;
            else
              p <= 0;
          end
      endcase
    end
  end
endmodule
