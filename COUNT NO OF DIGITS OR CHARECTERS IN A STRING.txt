COUNT NO OF DIGITS OR CHARECTERS IN A STRING:

module count_no_of_digits_or_charecters_in_a_string(clk,reset,in,out);
  input clk,reset;
  input string in;
  output reg [7:0]out;
  
  string in_;
  
  parameter size = 15;
  
  reg [7:0]count;
  
  int i;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      in_ <= {size{"0"}};
      count <= 0;
    end
    else
      begin
        for(i=0; i<size; i++) begin
          in_[i] = in[i];
          count = count+1;
          out <= count;
        end
        count <= 0;
      end
  end
endmodule
