module spy_number(clk,reset,num,out);
  input clk,reset;
  input [2:0]num;
  output reg [2:0]out;
  
  reg [2:0]dig_1,dig_2,result_1,result_2;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  parameter s3 = 2'b11;
  
  
  reg[1:0]current_state,next_state;
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
      begin
        next_state = s1;
      end
      s1:
      begin
        if(dig_1 > 0)
          next_state = s1;
        else
          next_state = s2;
      end
      s2:
      begin
        if(dig_2 > 0)
          next_state = s2;
        else
          next_state = s3;
      end
      s3:
        begin
          next_state = s0;
        end
    endcase
  end
      
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      dig_1 <= 0;
      dig_2 <= 0;
      current_state <= s0;
      result_1 <= 0;
      result_2 <= 1;
    end
    else
      begin
        current_state <= next_state;
        case(current_state)
            s0:
            begin
              out <= 0;
              dig_1 <= num;
              dig_2 <= num;
            end
            s1:
            begin
              if( dig_1 > 0) begin
                result_1 <= (dig_1 % 10) + result_1;
                dig_1 <= dig_1 / 10;
              end
            end
            s2:
            begin
              if( dig_2 > 0) begin
                result_2 <= (dig_2 % 10) * result_2;
                dig_2 <= dig_2 / 10;
              end
            end
          s3:
            begin
              if (result_1 == result_2)
                out <= 1;
            end
        endcase
      end
  end
endmodule

  reg [2:0]num;
  wire [2:0]out;
  
  spy_number u1(.clk(clk),.reset(reset),.num(num),.out(out));
  
  initial
    begin
      clk = 0;
      forever #5 clk = ~clk;
    end
  
  initial
    begin
      reset = 0; #5;
      reset = 1; #2;
      num = 126;
      #100;
      $finish;
    end
  initial
    begin
      $monitor("time=%0t,clk=%0b,reset=%0b,num=%0d,out=%0b",$time,clk,reset,num,out);
    end
endmodule
