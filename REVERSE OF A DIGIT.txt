REVERSE OF A DIGIT

module reverse_of_a_digit(clk,reset,num,out);
  input clk,reset;
  input [7:0]num;
  output reg [7:0]out;
  
  reg [7:0]reverse,digit_1;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      reverse <= 0;
      digit_1 <= num;
    end
    else
      begin
        if( digit_1 > 0) begin
          reverse <= reverse * 10 + (digit_1 % 10);
          digit_1 <= digit_1 / 10;
        end
        else
          begin
            out <= reverse;
          end
      end
  end
endmodule
