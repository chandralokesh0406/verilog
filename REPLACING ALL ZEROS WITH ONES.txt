module replacing_all_zreos_with_ones(clk,reset,num,out);
  input clk,reset;
  input [7:0]num;
  output reg[7:0]out;
  
  int i;
  reg [7:0]digit;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 8'b0;
      digit <= num;
    end
    else
      begin
        for(i=0; i<8; i++) begin
          if(digit[i] == 0) begin
            digit[i] = 1;
            out <= digit;
          end
        end
      end
  end
endmodule
