design:
module checking_harshad_number(clk,reset,num,out);
  input clk,reset;
  input [7:0]num;
  output reg out;
  
  reg [7:0]temp,temp_1,sum;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      temp_1 <= num;
      temp <= 0;
      sum <= 0;
    end
      else if(temp_1 > 0) begin
        temp <= temp_1 % 10;
        temp_1 <= temp_1 / 10;
      end
    else
      sum <= sum + temp;
    if(num % sum == 0)
      out <= 1;
  end
endmodule
