🧾 1. Alternate Digit Sum
🔹 Objective

Add digits at even positions and subtract digits at odd positions

🧠 Concept

Digit extraction + indexing

🔍 Example

N = 134 → (1+4) - (3)

🎯 Output

👉 2


module problem_statement(clk,reset,num,out);
  input clk,reset;
  input [7:0]num;
  output reg [3:0]out;
  
  reg [7:0] count,digit,digit_1,sum,sum_1;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      count <= 0;
      digit <= num;
      digit_1 <= num;
      sum <= 0;
      sum_1 <= 0;
    end
    else if( digit > 0) begin
      digit <= digit / 10;
      if( count % 2 == 0)
        sum <= sum + (digit % 10);
      else
        sum_1 <= sum_1 + ( digit % 10) ;
      count <= count + 1;
    end
    else
      out <= sum - sum_1;
  end
endmodule
