design:

module digit_extraction_factorial(clk,reset,num,out);
  input clk,reset;
  input [15:0]num;
  output reg [15:0]out;
  
  
  reg[2:0]current_state,next_state;
  
  reg [15:0]digit,digit_1,digit_2,digit_3;
  
  parameter s0 = 3'b000;
  parameter s1 = 3'b001;
  parameter s2 = 3'b010;
  parameter s3 = 3'b011;
  parameter s4 = 3'b100;
  
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
        s0:
        begin
          next_state = s1;
        end
        s1:
        begin
          if( digit_1 > 0)
            next_state = s2;
          else
            next_state = s0;
        end
        s2:
          begin
            next_state = s3;
          end
        s3:
          begin
            if(digit_3 > 0)
              next_state = s3;
            else
              next_state = s4;
          end
        s4:
          begin
            next_state = s1;
          end
    endcase
  end

            
      
      
          
  
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      current_state <= s0;
      out <= 0;
      digit <= num;
      digit_1 <= 0;
      digit_2 <= 1;
      digit_3 <= 0;
    end
    else begin
      current_state <= next_state;
      case(current_state)
          s0:
          begin
            digit <= 0;
            digit_1 <= num;
            digit_2 <= 1;
            digit_3 <= 0;
            out <= 0;
          end
          s1:
          begin
            if(digit_1 > 0) begin
              digit <= digit_1 % 10;
              digit_1 <= digit_1 / 10;
            end
          end
          s2:
          begin
            digit_3 <= digit-1;
            digit_2 <= digit;
          end
          s3:
          begin
            if(digit_3 > 0) begin
              digit_3 <= digit_3-1;
              digit_2 <= digit_2 * digit_3;
            end
          end
          s4:
            begin
              out <= out + digit_2;
            end
      endcase
    end
  end
endmodule

  initial

    end

    end
endmodule
