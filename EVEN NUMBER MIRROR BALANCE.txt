EVEN NUMBER MIRROR BALANCE:

module even_number_mirror_balance(clk,reset,num,out);
  input clk,reset;
  input [31:0]num;
  output reg out;
  
  reg [31:0]digit,digit_1;
  reg [3:0]out_1,out_2;
  reg [3:0]count,count_1;
  
  always @(*) begin
    digit_1 = num;
    count = 0;
  while (digit_1 > 0) begin
    digit_1  = digit_1 / 10;
    count = count+1;
  end
  end
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      count_1 <= 0;
      digit <= num;
      out_1 <= 0;
      out_2 <= 0;
    end
      
    else
      begin
        if(count_1 != count/2) begin
        digit <= digit / 10;
        out_1 <= out_1 + (digit%10);
        count_1 <= count_1+1;
        end
        else if( digit > 0) begin
          digit <= digit / 10;
          out_2 <= out_2 +( digit % 10);
        end
        else
          if( out_1 == out_2)
            out <= 1;
      end
  end
endmodule
