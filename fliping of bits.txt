module fliping_of_bits(clk,reset,bits,out);
  input clk,reset;
  input [3:0]bits;
  output reg [3:0]out;
  
  reg [2:0]i;
  parameter size = 4;
  
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 4'b0000;
    else
      for(i=0; i<size; i++) begin
        if(i % 2 == 0)
          out[i] <= ~bits[i];
        else
          out[i] <= bits[i];
      end
  end
endmodule
