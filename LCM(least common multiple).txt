module least_common_multiple(clk,reset,num1,num2,out);
  input clk,reset;
  input [7:0]num1;
  input [7:0]num2;
  output reg [7:0]out;
  
  reg [7:0]i;
  reg [7:0]j;
  
  
  always @(*) begin
    if(num1 > num2)
      i = num1;
    else
      i = num2;
  end
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      j <= i;
    end
    else
      begin
        if( out == 0) begin
          if (j % num1 == 0 && j % num2 == 0)
            out <= j;
          else
            j <= j+1;
        end
      end
  end
endmodule
