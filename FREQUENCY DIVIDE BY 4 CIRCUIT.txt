FREQUENCY DIVIDE BY 4 CIRCUIT:

module frequency_divide_by_4_circuit(clk,reset,clock_out);
  input clk,reset;
  output reg clock_out;
  
  
  reg [1:0]count;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      clock_out <= 0;
      count <= 0;
    end
    else
      begin
        if(count == 2'd1) begin
          clock_out <= ~clock_out;
          count <= 0;
        end
        else
          count <= count + 1;
      end
  end
endmodule
