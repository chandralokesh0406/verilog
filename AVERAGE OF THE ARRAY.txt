AVERAGE OF THE ARRAY:

module finding_average_of_array(clk,reset,arr,out);
  input clk,reset;
  input [7:0]arr[0:4];
  output reg [7:0]out;
  
  parameter size = 5;
  
  reg [7:0]sum;
  
  reg [2:0]i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      i <= 0;
      sum <= 0;
    end
    else if(i < size)  begin
      sum <= sum + arr[i];
      i <= i+1;
    end
    else
      out <= sum/size;
  end
endmodule
