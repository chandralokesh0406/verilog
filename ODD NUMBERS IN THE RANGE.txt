ODD NUMBERS IN THE RANGE:

module odd_numbers_in_the_range(clk,reset,num1,num2,odd);
  input clk,reset;
  input  [127:0]num1,num2;
  output reg [127:0]odd;
  
  
  reg [127:0]num,digit;
  reg [1:0]i;
  
  parameter s0 = 2'b00;
  parameter s1 = 2'b01;
  parameter s2 = 2'b10;
  
  reg [1:0]current_state,next_state;
  
  
  always @(*) begin
    next_state = current_state;
    case(current_state)
      s0:
        begin
          next_state = s1;
        end
      s1:
        begin
          if( num < digit )
            next_state = s1;
          else
            next_state = s0;
        end
    endcase
  end
          
          
      
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      current_state <= s0;
      num <= num1;
      digit <= num2;
      odd <= 0;
      i <= 2;
    end
    else
      begin
        current_state <= next_state;
      case(current_state)
        s0:
          begin
            num <= num1;
            digit <= num2;
            odd <= 0;
            i <= 2;
          end
        s1:
          begin
            if( num % i != 0 && num < digit) begin
              odd <= odd + 1;
              num <= num+1;
            end
            else begin
              num <= num+1;
            end
          end
      endcase
      end
  end
Endmodule

  initial
    begin
      clk = 0;
      forever #5 clk = ~clk;
    end
  
  initial
    begin
      reset = 0;
      num1 = 10;
      num2 = 20;
      #10;
      reset = 1;
      #140;
      $finish;
    end
  initial
    begin
      $monitor("time=%0t,clk=%0b,reset=%0b,num1=%0d,num2=%0d,odd=%0d",$time,clk,reset,num1,num2,odd);
    end
endmodule
