module fibonacci_series_using_an_array(clk,reset,arr,out);
  input clk,reset;
  input [7:0]arr[0:4];
  output reg [7:0]out;
  
  reg [7:0]arr_num;
  
  parameter size = 5;
  
  int i;
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      arr_num <= 0;
      i <= 0;
    end
    else
      begin
        if ( i <= size) begin
          out <= out + arr_num;
          arr_num <= arr[i];
          i <= i+1;
        end
      end
  end
endmodule
