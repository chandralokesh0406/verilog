MULTIPLICATION TABLE:

module multiplication_table(clk,reset,num,out);
  input clk,reset;
  input [7:0]num;
  output reg [7:0] out;
  
  
  int i;
  
  always @(posedge clk or negedge reset) begin
    if(!reset)
      out <= 0;
    else
      begin
        if( i<13) begin
          out <= num * i;
          i <= i+1;
        end
      end
  end
endmodule
