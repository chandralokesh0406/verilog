module GCD(clk,reset,num1,num2,out);
  input clk,reset;
  input [7:0]num1,num2;
  output reg [7:0]out;
  
  reg [7:0]a,b;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= 0;
      a <= num1;
      b <= num2;
    end
    else if(a != b) begin
      if(a > b)
        a <= a-b;
      else
        b <= b-a;
    end
    else
      out = a;
  end
endmodule
