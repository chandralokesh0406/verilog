 RECTANGULAR STAR PATTERN:

module rectangular_star_pattern(clk,reset);
  input clk,reset;
  
  
  reg [3:0]row,col;
  
  parameter size = 9;
  parameter size_1 = 5;
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      col <= 0;
      row <= 0;
    end
    else
      begin
        if(row < size_1) begin
          if (col < size - 1) begin
          $write(" * ");
          col <= col + 1;
        end
        else begin
          $write(" * ");
          $display("");
          row <= row + 1;
          col <= 0;
        end
        end
      end
  end
endmodule

tb:

module tb;
  reg clk,reset;
  
  
  rectangular_star_pattern  u1(.clk(clk),.reset(reset));
  
  
  initial
    begin
      clk = 0;
      forever #5 clk = ~clk;
    end
  
  initial
    begin
      reset = 0;
      #5;
      reset = 1;
      
      #900;
      $finish;
    end
endmodule
