SRAM DESIGN:

module SRAM(clk,reset,wr_en,addr,data_in,data_out);
  input clk,reset;
  input wr_en;
  input [7:0]data_in;
  input [2:0]addr;
  output reg [7:0]data_out;
  reg [7:0]mem[0:7];
  
  
  always @(posedge clk or negedge reset) begin
    if(!reset) begin
      data_out <= 0;
    end
    else
      begin
        if(wr_en == 1)
          mem[addr] <= data_in;
        else
          data_out <= mem[addr];
      end
  end
endmodule
