module reverse_a_array(clk,reset,in,out);
  input clk,reset;
  input [7:0]in[0:4];
  output reg [7:0]out[0:4];
  
  reg [7:0]i;
  
  parameter size = 5;

  always  @(posedge clk or negedge reset) begin
    if(!reset) begin
      out <= {0,0,0,0,0};
      i <= 0;
    end
    else
      begin
        for(i=0; i<size; i++) begin
          out[i] = in[size-i-1];
        end
      end
  end
endmodule
